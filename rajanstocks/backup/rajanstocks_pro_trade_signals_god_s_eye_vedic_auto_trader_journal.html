// 1. Z-TREND (RZT 14, 2, 5, 1) SIGNAL EXTRACTOR (Moves chart-only Buy/Sell/Zones/SL into Signal Engine)
function computeZTrendSignal(candles, params = { len: 14, mult: 2, zoneLookback: 5 }) {
  const last = candles[candles.length - 1];
  const prev = candles[candles.length - 2] || last;
  const atr = candles.slice(-params.len).reduce((acc, c) => acc + (c.high - c.low), 0) / params.len;
  const upperBand = last.close + atr * params.mult;
  const lowerBand = last.close - atr * params.mult;
  const trendDir = last.close >= prev.close ? 'BUY' : 'SELL';
  return {
    indicator: 'Z_TREND',
    signal: trendDir,
    entryPrice: +(last.close).toFixed(2),
    trailingSL: +(trendDir === 'BUY' ? lowerBand : upperBand).toFixed(2),
    zone1Resistance: +(last.close + atr * 1.8).toFixed(2),
    zone2Resistance: +(last.close + atr * 3.1).toFixed(2),
    zone1Support: +(last.close - atr * 1.4).toFixed(2),
    confidence: 0.84
  };
}

// 2. AUTO-TRADER TIMING (9:15 BUY / 9:20 OPTION SELL) & ₹30,000 DAILY P&L TRAILING GUARD
function evaluateAutoTradeGuard({ side, instrumentType, currentIstHHMM, dailyTotalPnl, peakDailyPnl, activeIndicators }) {
  // Rule A: Option Buying & Stocks start at 09:15; Option Selling (writing) starts strictly at 09:20
  if (currentIstHHMM < '09:15') return { allowed: false, reason: 'PRE_MARKET_LOCKED' };
  if (instrumentType === 'OPTION' && side === 'SELL' && currentIstHHMM < '09:20') {
    return { allowed: false, reason: 'OPTION_SELL_LOCKED_UNTIL_09_20' };
  }

  // Rule B: Fix indicator bug -> Evaluate ALL active indicators including Z_TREND & TIME_CYCLE
  const requiredNames = ['Z_TREND', 'TIME_CYCLE', 'VWAP', 'EMA_20', 'RSI_14', 'MACD', 'OI_FLOW', 'VEDIC_GANN'];
  const agreed = requiredNames.filter(name => activeIndicators[name]?.agreesWith(side));
  const consensusRatio = agreed.length / requiredNames.length;

  // Rule C: ₹30,000 Daily P&L Limit -> Activate Profit Trailing & Throttle Trade Frequency
  const absPnl = Math.abs(dailyTotalPnl);
  if (absPnl >= 30000) {
    const trailingStopPnl = Math.max(30000 * 0.85, peakDailyPnl * 0.85);
    if (dailyTotalPnl > 0 && dailyTotalPnl <= trailingStopPnl) {
      return { allowed: false, mode: 'TRAILING_STOP_LOCKED', trailingFloor: trailingStopPnl };
    }
    // Require higher indicator agreement (>= 6/8) and max 1 open trade when >= ₹30,000
    return {
      allowed: agreed.length >= 6 && activeIndicators.Z_TREND?.agreesWith(side),
      mode: 'THROTTLED_TRAILING_30K',
      maxConcurrentTrades: 1,
      trailingFloor: trailingStopPnl,
      agreedIndicators: agreed
    };
  }

  return {
    allowed: agreed.length >= 5 && activeIndicators.Z_TREND?.agreesWith(side),
    mode: 'STANDARD_EXECUTION',
    maxConcurrentTrades: 4,
    agreedIndicators: agreed,
    consensusRatio
  };
}

// 3. JOURNAL TRADE DURATION & TIMESTAMP FORMATTER
function formatTradeTimestamps(entryMs, exitMs = null) {
  const fmt = (ms) => new Date(ms).toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: true });
  const endMs = exitMs || Date.now();
  const diffSec = Math.max(0, Math.floor((endMs - entryMs) / 1000));
  const hrs = Math.floor(diffSec / 3600);
  const mins = Math.floor((diffSec % 3600) / 60);
  const secs = diffSec % 60;
  return {
    entryFormatted: fmt(entryMs),
    exitFormatted: exitMs ? fmt(exitMs) : 'LIVE (Open)',
    holdDuration: hrs > 0 ? `${hrs}h ${mins}m ${secs}s` : `${mins}m ${secs}s`
  };
}
